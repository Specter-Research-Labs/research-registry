\begin{document}
\begin{tikzpicture}[
    every node/.style={font=\small}
]

% Left: Wild-type
\begin{scope}[shift={(-6.0, 0)}]
    \node[paneltitle, align=center] at (0, 1.1) {Wild-type\\(ds\_0124\_thm\_2218)};

    \node[goal, minimum width=5.0cm] (w0) at (0, 0.2) {Goal: $N_7 - N_7^{distinct} = 3600$};
    \node[goal, below=1.55cm of w0, minimum width=5.0cm] (w1) {Subgoal: $N_7 - N_7^{distinct} = 3600$};
    \node[goal, below=1.55cm of w1, minimum width=3.2cm] (w2) {$\mathsf{False}$};
    \node[terminal, below=1.55cm of w2, minimum width=3.2cm] (w3) {\Large$\bullet$};

    \draw[divergetactic] (w0) -- node[midway, left, tacticlabel] {\texttt{intros}} (w1);
    \draw[tactic] (w1) -- node[midway, left, tacticlabel] {\texttt{norm\_num1}} (w2);
    \draw[tactic] (w2) -- node[midway, left, tacticlabel] {\texttt{norm\_num at *}} (w3);
\end{scope}

% Right: Intervention
\begin{scope}[shift={(6.0, 0)}]
    \node[paneltitle, align=center] at (0, 1.1) {Intervention\\(\texttt{block intros})};

    \node[divergenode, minimum width=5.0cm] (i0) at (0, 0.2) {Goal: $N_7 - N_7^{distinct} = 3600$};
    \node[goal, below=1.55cm of i0, minimum width=5.0cm] (i1) {Alt. subgoal: $N_7^{distinct} = 840 \to \dots \to \Delta = 3600$};
    \node[goal, below=1.55cm of i1, minimum width=4.2cm] (i2) {$\mathsf{False} \to \mathsf{True} \to \mathsf{False}$};
    \node[terminal, below=1.55cm of i2, minimum width=3.2cm] (i3) {\Large$\bullet$};
    \node[dead, right=3.0cm of i0, minimum width=2.7cm] (i_dead) {blocked};

    \draw[divergeblock] (i0.east) -- node[pos=0.40, above=5pt, tacticlabel, text=blocked, fill=white, inner sep=1pt] {block \texttt{intros}} (i_dead.west);
    \draw[divergetactic] (i0) -- node[midway, left, tacticlabel] {\texttt{intro $N_7^{distinct}$}} (i1);
    \draw[tactic] (i1) -- node[midway, left, tacticlabel] {\texttt{norm\_num1}} (i2);
    \draw[tactic] (i2) -- node[midway, left, tacticlabel] {\texttt{tauto}} (i3);
\end{scope}

% Crosslink: same goal
\draw[crosslink] (w0.east) -- (i0.west) node[midway, above, annotbg] {same goal};

\end{tikzpicture}
