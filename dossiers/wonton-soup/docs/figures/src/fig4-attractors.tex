% Attractor Analysis: GED Matrix, Clustering, Basins
\documentclass[tikz,border=5pt]{standalone}

\input{common-styles.tex}

\begin{document}
\begin{tikzpicture}[
    every node/.style={font=\small},
    show background rectangle,
    background rectangle/.style={fill=white}
]

\tikzset{
    cell/.style={draw=black!15, line width=0.4pt}, % Lighter grid lines
    dendro/.style={draw=muted, line width=0.8pt, rounded corners=1pt}, % Smoother dendrogram
    faint/.style={draw=muted!50}
}

% Panel A: GED Matrix
\begin{scope}[shift={(0, 0)}]
    \node[paneltitle, anchor=south] at (1.5, 3.5) {(A) GED matrix};
    \def\cellsize{0.6}

    % Matrix cells
    \foreach \i in {0,1,2,3,4} {
        \foreach \j in {0,1,2,3,4} {
            \ifnum\i=\j
                \fill[white] (\j*\cellsize, -\i*\cellsize) rectangle +(\cellsize, \cellsize);
                \node[font=\scriptsize, text=ink] at (\j*\cellsize + 0.5*\cellsize, -\i*\cellsize + 0.5*\cellsize) {0};
            \else
                % Fake data for illustration
                \pgfmathtruncatemacro{\dist}{abs(\i-\j)}
                \pgfmathtruncatemacro{\val}{\dist * 2 + 1}
                \pgfmathsetmacro{\shade}{15 + \dist*18}
                \fill[heatbase!\shade!white] (\j*\cellsize, -\i*\cellsize) rectangle +(\cellsize, \cellsize);
                \node[font=\scriptsize, text=ink] at (\j*\cellsize + 0.5*\cellsize, -\i*\cellsize + 0.5*\cellsize) {\val};
            \fi
            \draw[cell] (\j*\cellsize, -\i*\cellsize) rectangle +(\cellsize, \cellsize);
        }
    }

    % Row labels
    \foreach \i/\lbl in {0/wild,1/blk-s,2/blk-r,3/blk-i,4/ctrl} {
        \node[anchor=east, font=\scriptsize, align=right] at (-0.08, -\i*\cellsize + 0.5*\cellsize) {\lbl};
    }

    % Column labels (abbreviated)
    \foreach \j/\lbl in {0/w,1/s,2/r,3/i,4/c} {
        \node[anchor=south, font=\scriptsize] at (\j*\cellsize + 0.5*\cellsize, \cellsize + 0.05) {\lbl};
    }

    % Colorbar
    \begin{scope}[shift={(3.5, -1.8)}]
        \foreach \k in {0,1,...,10} {
            \pgfmathsetmacro{\shade}{15 + \k*8}
            \fill[heatbase!\shade!white] (\k*0.15, 0) rectangle +(\k*0.15 + 0.15, 0.25);
        }
        % Continuous gradient simulation
        \shade[left color=heatbase!15!white, right color=heatbase!95!white] (0,0) rectangle (1.65, 0.25);
        \draw[black!20] (0, 0) rectangle (1.65, 0.25);
        
        \node[anchor=north, font=\tiny, text=muted] at (0, -0.05) {low};
        \node[anchor=north, font=\tiny, text=muted] at (1.65, -0.05) {high};
        \node[anchor=west, font=\scriptsize, text=muted] at (1.8, 0.12) {GED};

        \node[annot, anchor=west, text width=2.5cm, align=left] at (-0.5, -0.8)
            {w: wild \quad s: blk-s\\r: blk-r \quad i: blk-i\\c: ctrl};
    \end{scope}
\end{scope}

% Panel B: Dendrogram
\begin{scope}[shift={(8.0, 0)}]
    \node[paneltitle, anchor=south] at (1.6, 3.5) {(B) Clustering};

    % Tree structure
    % Root at (1.6, 0) approx high up
    % Structure: ((wild, ctrl), ((blk_s, blk_r), blk_i)) -- made up structure

    % Coordinates
    \coordinate (wild) at (0, -2.8);
    \coordinate (ctrl) at (0.8, -2.8);
    \coordinate (blks) at (1.6, -2.8);
    \coordinate (blkr) at (2.4, -2.8);
    \coordinate (blki) at (3.2, -2.8);

    % Labels
    \node[anchor=north, font=\scriptsize] at (wild) {wild};
    \node[anchor=north, font=\scriptsize] at (ctrl) {ctrl};
    \node[anchor=north, font=\scriptsize] at (blks) {blk-s};
    \node[anchor=north, font=\scriptsize] at (blkr) {blk-r};
    \node[anchor=north, font=\scriptsize] at (blki) {blk-i};

    % Edges - squared
    % wild-ctrl merge
    \draw[dendro] (wild) -- (wild |- 0, -1.5) -- (ctrl |- 0, -1.5) -- (ctrl);
    \coordinate (m1) at (0.4, -1.5);
    \fill[ink] (m1) circle (1.5pt);

    % blks-blkr merge
    \draw[dendro] (blks) -- (blks |- 0, -2.0) -- (blkr |- 0, -2.0) -- (blkr);
    \coordinate (m2) at (2.0, -2.0);
    \fill[ink] (m2) circle (1.5pt);

    % m2 - blki merge
    \draw[dendro] (m2) -- (m2 |- 0, -1.0) -- (blki |- 0, -1.0) -- (blki);
    \coordinate (m3) at (2.6, -1.0);
    \fill[ink] (m3) circle (1.5pt);

    % m1 - m3 merge (Root)
    \draw[dendro] (m1) -- (m1 |- 0, -0.2) -- (m3 |- 0, -0.2) -- (m3);
    \coordinate (root) at (1.5, -0.2);
    \fill[ink] (root) circle (1.5pt);

    % Distance axis
    \draw[faint, ->] (-0.6, -2.8) -- (-0.6, 0.0);
    \node[anchor=south, font=\scriptsize, rotate=90, text=muted] at (-0.8, -1.4) {Merge Distance};
    
    % Ticks
    \foreach \y/\l in {-2.8/0, -1.5/5, -0.2/10} {
        \draw[faint] (-0.65, \y) -- (-0.55, \y);
    }
    
    % Cut line
    \draw[dashed, blocked, line width=0.6pt] (-0.4, -0.8) -- (3.6, -0.8);
    \node[annot, text=blocked, anchor=west] at (3.65, -0.8) {cut};
\end{scope}

% Panel C: Basins of attraction
\begin{scope}[shift={(0, -9.0)}]
    \node[paneltitle, anchor=south] at (4.0, 3.5) {(C) Basins of attraction};

    % Basin 1 (Topological Contours)
    \foreach \r/\o in {1.6/0.15, 1.3/0.3, 1.0/0.45, 0.7/0.6} {
        \pgfmathsetmacro{\ry}{\r*0.875}
        \fill[basin1, opacity=\o] (1.5, 1.2) ellipse (\r cm and \ry cm);
        \draw[basin1!80!black, line width=0.2pt, opacity=0.5] (1.5, 1.2) ellipse (\r cm and \ry cm);
    }
    \node[font=\bfseries, text=ink] at (1.5, -0.6) {Basin 1};
    \node[annot] at (1.5, -1.0) {(dominant)};
    \fill[ink] (1.5, 1.0) circle (3pt) node[right, xshift=2pt, font=\scriptsize] {Attractor A};

    % Basin 2 (Topological Contours)
    \foreach \r/\o in {1.4/0.15, 1.1/0.3, 0.8/0.45, 0.5/0.6} {
        \pgfmathsetmacro{\ry}{\r*0.85}
        \fill[basin2, opacity=\o] (4.8, 0.8) ellipse (\r cm and \ry cm);
        \draw[basin2!80!black, line width=0.2pt, opacity=0.5] (4.8, 0.8) ellipse (\r cm and \ry cm);
    }
    \node[font=\bfseries, text=ink] at (4.8, -0.6) {Basin 2};
    \fill[ink] (4.8, 0.6) circle (3pt) node[right, xshift=2pt, font=\scriptsize] {Attractor B};

    % Basin 3 (Topological Contours)
    \foreach \r/\o in {1.1/0.15, 0.85/0.3, 0.6/0.45, 0.35/0.6} {
        \pgfmathsetmacro{\ry}{\r*0.9}
        \fill[basin3, opacity=\o] (7.5, 1.5) ellipse (\r cm and \ry cm);
        \draw[basin3!80!black, line width=0.2pt, opacity=0.5] (7.5, 1.5) ellipse (\r cm and \ry cm);
    }
    \node[font=\bfseries, text=ink] at (7.5, -0.6) {Basin 3};
    \fill[ink] (7.5, 1.5) circle (3pt) node[right, xshift=2pt, font=\scriptsize] {Attractor C};

    % Trajectories
    % Basin 1 trajectories
    \foreach \ang in {45, 135, 225, 315} {
        \draw[trajectory, ->] ($(1.5, 1.0) + (\ang:1.2)$) to[out=\ang+180, in=\ang] (1.5, 1.0);
        \fill[black!40] ($(1.5, 1.0) + (\ang:1.2)$) circle (1.5pt);
    }
    % Basin 2 trajectories
    \foreach \ang in {60, 180, 300} {
         \draw[trajectory, ->] ($(4.8, 0.6) + (\ang:1.0)$) to[out=\ang+160, in=\ang] (4.8, 0.6);
         \fill[black!40] ($(4.8, 0.6) + (\ang:1.0)$) circle (1.5pt);
    }
    % Basin 3 trajectories
    \foreach \ang in {90, 270} {
         \draw[trajectory, ->] ($(7.5, 1.5) + (\ang:0.8)$) to[out=\ang+180, in=\ang] (7.5, 1.5);
         \fill[black!40] ($(7.5, 1.5) + (\ang:0.8)$) circle (1.5pt);
    }

    % Stats
    \node[annotbg, text=ink] at (1.5, 2.3) {45\% seeds};
    \node[annotbg, text=ink] at (4.8, 1.8) {36\% seeds};
    \node[annotbg, text=ink] at (7.5, 2.3) {18\% seeds};

    % Legend
    \begin{scope}[shift={(10.5, 1.0)}]
        \fill[black!40] (0, 0) circle (1.5pt);
        \node[anchor=west, font=\scriptsize] at (0.2, 0) {Seed state};

        \fill[ink] (0, -0.4) circle (3pt);
        \node[anchor=west, font=\scriptsize] at (0.2, -0.4) {Attractor};

        \draw[trajectory, ->] (0, -0.8) -- (0.5, -0.8);
        \node[anchor=west, font=\scriptsize] at (0.6, -0.8) {Trajectory};
    \end{scope}

\end{scope}

\end{tikzpicture}
