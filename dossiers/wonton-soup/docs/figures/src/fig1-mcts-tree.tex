            node[dead] (n1a) {$\ldots$}

            node[terminal, label={[visit]below:{$N=4$}}] (n1b) {\Large$\bullet$}

    }
    child {
        node[goal, label={[visit]above right:{$N=5, \mathbf{Q=0.85}$}}] (n2) {$a*b+a*c=\ldots$}

% Selection path highlight (Orange Arrow)
